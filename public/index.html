const express = require('express'); const path = require('path'); const dotenv = require('dotenv'); const { db, uuid, logAudit } = require('./db'); const { parseQuantity, formatQuantity } = require('./utils/fraction'); const { calculateDailySnapshot } = require('./services/inventoryEngine'); dotenv.config(); const app = express(); const PORT = Number(process.env.PORT || 4000); const publicDir = path.join(__dirname, '../public'); app.use(express.json({ limit: '2mb' })); app.use(express.static(publicDir)); function requireValidQuantity(input, label) { const parsed = parseQuantity(input); if (parsed === null || Number.isNaN(parsed) || parsed < 0) { throw new Error(`Invalid ${label}. Please enter a whole number or fraction such as 1/2, 3/4, or 2 1/2.`); } return parsed; } function ensureItemExists(itemId) { const item = db.prepare('SELECT * FROM items WHERE id = ?').get(itemId); if (!item) { throw new Error('Unknown item. Please select a valid inventory item.'); } return item; } function getPreviousPhysicalEnding(itemId, operatingDate) { const row = db.prepare(` SELECT quantity FROM physical_counts WHERE item_id = ? AND operating_date < ? ORDER BY date(operating_date) DESC, created_at DESC LIMIT 1 `).get(itemId, operatingDate); return row ? Number(row.quantity) : 0; } function getDailySummaryForItem(itemId, operatingDate) { const item = ensureItemExists(itemId); const beginning = getPreviousPhysicalEnding(itemId, operatingDate); const production = db.prepare(` SELECT COALESCE(SUM(quantity), 0) AS total FROM production_transactions WHERE item_id = ? AND operating_date = ? `).get(itemId, operatingDate)?.total || 0; const stockIn = db.prepare(` SELECT COALESCE(SUM(quantity), 0) AS total FROM stock_in_transactions WHERE item_id = ? AND operating_date = ? `).get(itemId, operatingDate)?.total || 0; const endingRow = db.prepare(` SELECT quantity FROM physical_counts WHERE item_id = ? AND operating_date = ? AND shift = 'NIGHT' ORDER BY created_at DESC LIMIT 1 `).get(itemId, operatingDate); const ending = endingRow ? Number(endingRow.quantity) : 0; const totalAvailable = beginning + Number(production) + Number(stockIn); const usage = totalAvailable - ending; return { itemId: item.id, itemName: item.name, beginning, production: Number(production), stocksIn: Number(stockIn), totalAvailable, physicalEnding: ending, actualUsage: usage, status: ending > totalAvailable ? 'CHECK COUNT — ENDING STOCK EXCEEDS AVAILABLE STOCK' : (ending <= 0 ? 'OUT OF STOCK' : (ending <= Number(item.critical_level || 0) ? 'CRITICAL LOW' : (ending <= Number(item.reorder_level || 0) ? 'LOW STOCK' : 'OK'))), suggestion: ending > totalAvailable ? 'MANAGEMENT REVIEW REQUIRED' : (ending <= 0 ? 'REPLENISH IMMEDIATELY' : (ending <= Number(item.critical_level || 0) ? 'REPLENISH IMMEDIATELY' : (ending <= Number(item.reorder_level || 0) ? 'REPLENISH SOON' : 'NO ACTION'))), recommendedReplenishment: Math.max(0, Number(item.target_stock || 0) - ending) }; } app.get('/api/health', (req, res) => { res.json({ ok: true, message: 'THC inventory server is running.' }); }); app.get('/api/items', (req, res) => { const rows = db.prepare('SELECT * FROM items ORDER BY name ASC').all(); res.json(rows.map((row) => ({ ...row, active: !!row.active, pending_review: !!row.pending_review }))); }); app.post('/api/items', (req, res) => { try { const { name, category, unit, reorder_level, critical_level, target_stock, active, notes } = req.body || {}; if (!name || !String(name).trim()) { return res.status(400).json({ error: 'Item name is required.' }); } const itemId = uuid(); const now = new Date().toISOString(); const record = { id: itemId, name: String(name).trim(), category: category || 'General', unit: unit || 'pcs', reorder_level: Number(reorder_level || 0), critical_level: Number(critical_level || 0), target_stock: Number(target_stock || 0), active: active === false ? 0 : 1, notes: notes || '', pending_review: 0, source: 'MANUAL', created_at: now, updated_at: now }; db.prepare(` INSERT INTO items (id, name, category, unit, reorder_level, critical_level, target_stock, active, notes, pending_review, source, created_at, updated_at) VALUES (@id, @name, @category, @unit, @reorder_level, @critical_level, @target_stock, @active, @notes, @pending_review, @source, @created_at, @updated_at) `).run(record); logAudit('items', itemId, 'create', record); res.status(201).json(record); } catch (error) { res.status(400).json({ error: error.message || 'Unable to create item.' }); } }); app.put('/api/items/:id', (req, res) => { const itemId = req.params.id; const current = ensureItemExists(itemId); const update = { ...current, ...req.body, updated_at: new Date().toISOString() }; db.prepare(` UPDATE items SET name = @name, category = @category, unit = @unit, reorder_level = @reorder_level, critical_level = @critical_level, target_stock = @target_stock, active = @active, notes = @notes, pending_review = @pending_review, updated_at = @updated_at WHERE id = @id `).run(update); logAudit('items', itemId, 'update', update); res.json(update); }); app.get('/api/production', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const rows = db.prepare(` SELECT p.*, i.name AS item_name FROM production_transactions p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? ORDER BY p.created_at DESC `).all(date); res.json(rows); }); app.post('/api/production', (req, res) => { try { const { item_id, itemId, date, operating_date, quantity, unit, notes } = req.body || {}; const targetItemId = item_id || itemId; const operatingDate = date || operating_date || new Date().toISOString().slice(0, 10); const parsedQty = requireValidQuantity(quantity, 'quantity'); const item = ensureItemExists(targetItemId); const transactionId = uuid(); const record = { id: transactionId, item_id: item.id, quantity: parsedQty, unit: unit || item.unit || 'pcs', operating_date: operatingDate, source: 'PRODUCTION TEAM', notes: notes || '', created_at: new Date().toISOString() }; db.prepare(` INSERT INTO production_transactions (id, item_id, quantity, unit, operating_date, source, notes, created_at) VALUES (@id, @item_id, @quantity, @unit, @operating_date, @source, @notes, @created_at) `).run(record); db.prepare(` INSERT INTO inventory_transactions (id, item_id, transaction_type, operating_date, quantity, unit, source, notes, created_at) VALUES (@id, @item_id, @transaction_type, @operating_date, @quantity, @unit, @source, @notes, @created_at) `).run({ id: uuid(), item_id: item.id, transaction_type: 'PRODUCTION', operating_date: operatingDate, quantity: parsedQty, unit: record.unit, source: 'PRODUCTION TEAM', notes: notes || '', created_at: new Date().toISOString() }); logAudit('production_transactions', transactionId, 'create', record); res.status(201).json(record); } catch (error) { res.status(400).json({ error: error.message || 'Unable to save production record.' }); } }); app.get('/api/stocks-in', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const rows = db.prepare(` SELECT s.*, i.name AS item_name FROM stock_in_transactions s LEFT JOIN items i ON i.id = s.item_id WHERE s.operating_date = ? ORDER BY s.created_at DESC `).all(date); res.json(rows); }); app.post('/api/stocks-in', (req, res) => { try { const { item_id, itemId, date, operating_date, quantity, unit, source, reference, notes } = req.body || {}; const targetItemId = item_id || itemId; const operatingDate = date || operating_date || new Date().toISOString().slice(0, 10); const parsedQty = requireValidQuantity(quantity, 'quantity'); const item = ensureItemExists(targetItemId); const transactionId = uuid(); const record = { id: transactionId, item_id: item.id, quantity: parsedQty, unit: unit || item.unit || 'pcs', operating_date: operatingDate, source: source || 'PURCHASE', reference: reference || '', notes: notes || '', created_at: new Date().toISOString() }; db.prepare(` INSERT INTO stock_in_transactions (id, item_id, quantity, unit, operating_date, source, reference, notes, created_at) VALUES (@id, @item_id, @quantity, @unit, @operating_date, @source, @reference, @notes, @created_at) `).run(record); db.prepare(` INSERT INTO inventory_transactions (id, item_id, transaction_type, operating_date, quantity, unit, source, notes, created_at) VALUES (@id, @item_id, @transaction_type, @operating_date, @quantity, @unit, @source, @notes, @created_at) `).run({ id: uuid(), item_id: item.id, transaction_type: 'STOCKS_IN', operating_date: operatingDate, quantity: parsedQty, unit: record.unit, source: record.source, notes: record.notes, created_at: new Date().toISOString() }); logAudit('stock_in_transactions', transactionId, 'create', record); res.status(201).json(record); } catch (error) { res.status(400).json({ error: error.message || 'Unable to save stock in record.' }); } }); app.get('/api/physical-counts', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const rows = db.prepare(` SELECT p.*, i.name AS item_name FROM physical_counts p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? ORDER BY i.name ASC `).all(date); res.json(rows); }); app.post('/api/physical-counts', (req, res) => { try { const { item_id, itemId, date, operating_date, quantity, notes, shift } = req.body || {}; const targetItemId = item_id || itemId; const operatingDate = date || operating_date || new Date().toISOString().slice(0, 10); const parsedQty = requireValidQuantity(quantity, 'quantity'); const item = ensureItemExists(targetItemId); const selectedShift = shift || 'NIGHT'; const existing = db.prepare('SELECT * FROM physical_counts WHERE item_id = ? AND operating_date = ? AND shift = ?').get(item.id, operatingDate, selectedShift); if (existing) { return res.status(409).json({ error: 'Duplicate physical count detected for the same date and shift. Please edit the existing count instead of creating another one.' }); } const record = { id: uuid(), item_id: item.id, operating_date: operatingDate, shift: selectedShift, quantity: parsedQty, notes: notes || '', created_at: new Date().toISOString(), updated_at: new Date().toISOString() }; db.prepare(` INSERT INTO physical_counts (id, item_id, operating_date, shift, quantity, notes, created_at, updated_at) VALUES (@id, @item_id, @operating_date, @shift, @quantity, @notes, @created_at, @updated_at) `).run(record); db.prepare(` INSERT INTO inventory_transactions (id, item_id, transaction_type, operating_date, quantity, unit, source, notes, created_at) VALUES (@id, @item_id, @transaction_type, @operating_date, @quantity, @unit, @source, @notes, @created_at) `).run({ id: uuid(), item_id: item.id, transaction_type: 'PHYSICAL_COUNT', operating_date: operatingDate, quantity: parsedQty, unit: item.unit, source: 'NIGHT OPERATIONS', notes: notes || '', created_at: new Date().toISOString() }); logAudit('physical_counts', record.id, 'create', record); res.status(201).json(record); } catch (error) { res.status(400).json({ error: error.message || 'Unable to save physical count.' }); } }); app.get('/api/dashboard', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const items = db.prepare('SELECT * FROM items WHERE active = 1').all(); const totalItems = items.length; let outOfStock = 0; let criticalLow = 0; let lowStock = 0; const lowStockItems = []; const replenishment = []; const recentActivity = db.prepare(` SELECT 'PRODUCTION' AS type, p.operating_date AS date, i.name AS item_name, p.quantity AS quantity, p.unit AS unit, p.source AS source FROM production_transactions p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? UNION ALL SELECT 'STOCKS_IN', s.operating_date, i.name, s.quantity, s.unit, s.source FROM stock_in_transactions s LEFT JOIN items i ON i.id = s.item_id WHERE s.operating_date = ? UNION ALL SELECT 'PHYSICAL_COUNT', p.operating_date, i.name, p.quantity, i.unit, 'NIGHT OPERATIONS' FROM physical_counts p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? ORDER BY date DESC, type ASC LIMIT 20 `).all(date, date, date); let productionTotal = 0; let stockInTotal = 0; let usageTotal = 0; const productionRows = db.prepare('SELECT COALESCE(SUM(quantity), 0) AS total FROM production_transactions WHERE operating_date = ?').get(date)?.total || 0; const stockInRows = db.prepare('SELECT COALESCE(SUM(quantity), 0) AS total FROM stock_in_transactions WHERE operating_date = ?').get(date)?.total || 0; productionTotal = Number(productionRows); stockInTotal = Number(stockInRows); for (const item of items) { const summary = getDailySummaryForItem(item.id, date); if (summary.physicalEnding <= 0) outOfStock += 1; if (summary.physicalEnding > 0 && summary.physicalEnding <= Number(item.critical_level || 0)) criticalLow += 1; if (summary.physicalEnding > 0 && summary.physicalEnding <= Number(item.reorder_level || 0)) lowStock += 1; if (summary.physicalEnding <= Number(item.reorder_level || 0)) { lowStockItems.push({ name: item.name, quantity: formatQuantity(summary.physicalEnding), unit: item.unit, status: summary.status, suggestion: summary.suggestion }); } if (summary.recommendedReplenishment > 0) { replenishment.push({ name: item.name, recommended: summary.recommendedReplenishment, unit: item.unit, action: summary.suggestion }); } usageTotal += summary.actualUsage; } res.json({ date, totalItems, outOfStock, criticalLow, lowStock, totalProductionToday: productionTotal, totalStocksInToday: stockInTotal, totalActualUsageToday: usageTotal, itemsRequiringReplenishment: replenishment.length, lowStockItems, replenishment, recentActivity }); }); app.get('/api/reports/daily', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const items = db.prepare('SELECT * FROM items WHERE active = 1').all(); const rows = items.map((item) => getDailySummaryForItem(item.id, date)); res.json(rows); }); app.get('/api/reports/production', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const rows = db.prepare(` SELECT p.*, i.name AS item_name FROM production_transactions p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? ORDER BY i.name ASC, p.created_at DESC `).all(date); res.json(rows); }); app.get('/api/reports/night', (req, res) => { const date = req.query.date || new Date().toISOString().slice(0, 10); const rows = db.prepare(` SELECT p.*, i.name AS item_name, i.unit AS item_unit FROM physical_counts p LEFT JOIN items i ON i.id = p.item_id WHERE p.operating_date = ? ORDER BY i.name ASC `).all(date); const enriched = rows.map((row) => { const summary = getDailySummaryForItem(row.item_id, date); return { ...row, beginning: summary.beginning, production: summary.production, stocksIn: summary.stocksIn, totalAvailable: summary.totalAvailable, actualUsage: summary.actualUsage, status: summary.status, suggestion: summary.suggestion }; }); res.json(enriched); }); app.get('/api/item-history/:id', (req, res) => { const item = ensureItemExists(req.params.id); const rows = db.prepare(` SELECT operating_date, quantity, unit, transaction_type, source, notes FROM inventory_transactions WHERE item_id = ? ORDER BY operating_date DESC, created_at DESC `).all(item.id); res.json({ item, history: rows }); }); app.get('/api/export/:type', (req, res) => { const type = req.params.type; const types = { items: 'SELECT * FROM items', production: 'SELECT * FROM production_transactions', stock_in: 'SELECT * FROM stock_in_transactions', physical_counts: 'SELECT * FROM physical_counts', inventory_transactions: 'SELECT * FROM inventory_transactions' }; const sql = types[type]; if (!sql) { return res.status(404).json({ error: 'Export type not found.' }); } const rows = db.prepare(sql).all(); const headers = Object.keys(rows[0] || {}); const csv = [headers.join(',')]; for (const row of rows) { csv.push(headers.map((header) => JSON.stringify(row[header] ?? '')).join(',')); } res.setHeader('Content-Type', 'text/csv'); res.setHeader('Content-Disposition', `attachment; filename="${type}.csv"`); res.send(csv.join('\n')); }); app.get('*', (req, res) => { res.sendFile(path.join(publicDir, 'index.html')); }); app.listen(PORT, () => { console.log(`THC Inventory server running on http://localhost:${PORT}`); });
